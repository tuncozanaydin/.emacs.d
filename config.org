* Margins
#+begin_src emacs-lisp :tangle config.el
;; Fringes don't exist under -nw, so `set-fringe-mode' can't pad a terminal.
;; Use text-area margins (measured in columns) for TTY windows instead, and
;; clear them on graphical windows so they never stack with the fringe padding.
;; The choice is per-window by frame type, so a mixed GUI+TTY daemon is handled.
(defvar toa/tty-margin 3
  "Left/right margin, in columns, for windows in terminal frames.")

(defun toa/apply-tty-margins (&rest _)
  "Give terminal windows left/right margins; clear margins on graphical ones.
Exempt the splash buffer (it centers itself) and any visual-fill-column buffer
(that mode owns its window's margins to enforce the wrap width)."
  (walk-windows
   (lambda (win)
     (let ((buf (window-buffer win)))
       (cond
        ((and (boundp 'visual-fill-column-mode)
              (buffer-local-value 'visual-fill-column-mode buf)))
        ((or (display-graphic-p (window-frame win))
             (eq buf (get-buffer "*splash*")))
         (set-window-margins win 0 0))
        (t (set-window-margins win toa/tty-margin toa/tty-margin)))))
   nil t))

;; Apply on structural changes (splits, minibuffer opening) via the config-change
;; hook, but ALSO on buffer changes within a window: `find-file' reuses the
;; window, which does NOT fire the config-change hook, so without this the margin
;; would only appear on the next structural change and shift the text sideways.
;; The startup hook covers the very first buffer, whose initial display predates
;; both hooks.
(add-hook 'window-configuration-change-hook #'toa/apply-tty-margins)
(add-hook 'window-buffer-change-functions #'toa/apply-tty-margins)
(add-hook 'emacs-startup-hook #'toa/apply-tty-margins)
#+end_src

* Visual Wrapping
#+begin_src emacs-lisp :tangle config.el
;; Wrap long lines visually at column 120 without inserting real newlines, in
;; every buffer. visual-line-mode does the word-wrapping; visual-fill-column
;; pins the wrap point to `fill-column' (rather than the window edge) by padding
;; the window with a right margin. The underlying text on disk is untouched.
;; Because visual-fill-column drives the margins, it supersedes the plain TTY
;; margins in wrapped buffers (see `toa/apply-tty-margins').
(setq-default fill-column 120)

;; visual-fill-column puts all the slack in the right margin, leaving the text
;; flush against the left edge. Restore the left breathing room (the same
;; `toa/tty-margin' width as the plain terminal margins) by shifting that many
;; columns from the right margin to the left: `extra-text-width' is (LEFT . RIGHT)
;; SUBTRACTED from each margin, so a negative left widens the left margin while a
;; matching positive right keeps the wrap width pinned at `fill-column'.
(setq visual-fill-column-extra-text-width (cons (- toa/tty-margin) toa/tty-margin))

(use-package visual-fill-column
  :ensure t
  :init
  (global-visual-line-mode 1)
  (global-visual-fill-column-mode 1))
#+end_src

* Splash Screen
#+begin_src emacs-lisp :tangle config.el
(defvar toa/splash-text "Ξ  M  Λ  C  S")
(defvar toa/splash-svg-path "~/.emacs.d/blackhole-lines.svg")

;; Dim, low-contrast face for the terminal wordmark (inherits Modus's `shadow',
;; so it stays theme-appropriate in both GUI and terminal).
(defface toa/splash-face '((t :inherit shadow))
  "Face for the terminal splash wordmark.")

(defun toa/splash-buffer ()
  "Get or create the splash buffer."
  (get-buffer-create "*splash*"))

(defun toa/splash-insert-centered (text &optional face)
  "Insert TEXT centered in the current buffer, propertized with FACE if given."
  (let* ((width (window-body-width))
         (padding (max 0 (/ (- width (string-width text)) 2))))
    (insert (make-string padding ?\s)
            (if face (propertize text 'face face) text)
            "\n")))

;; Display height (in text rows) of the SVG, so vertical centering can account
;; for an image that occupies more rows than the single line of text it sits on.
(defvar-local toa/splash--svg-rows 0)

(defun toa/splash-insert-svg ()
  "Insert SVG image centered in the buffer."
  (when (and (display-graphic-p)
             (image-type-available-p 'svg)
             (file-readable-p (expand-file-name toa/splash-svg-path)))
    (let* ((svg-path (expand-file-name toa/splash-svg-path))
           (img-width 600) ;; Fixed width in pixels
           (img (create-image svg-path 'svg nil :width img-width))
           (img-size (image-size img t))
           (img-cols (max 1 (floor (/ (car img-size) (float (frame-char-width))))))
           (padding (max 0 (/ (- (window-body-width) img-cols) 2))))
      (setq toa/splash--svg-rows
            (ceiling (/ (cdr img-size) (float (frame-char-height)))))
      (insert (make-string padding ?\s))
      (insert-image img)
      (insert "\n\n"))))

(defun toa/splash-render ()
  "Render the splash screen, centered vertically and horizontally."
  (with-current-buffer (toa/splash-buffer)
    (let ((inhibit-read-only t))
      (erase-buffer)
      (setq-local truncate-lines t
                  cursor-type nil
                  mode-line-format nil
                  header-line-format nil)
      ;; The splash centers itself; keep the global wrap modes from adding
      ;; margins here, which would throw off that centering.
      (visual-line-mode -1)
      (when (bound-and-true-p visual-fill-column-mode)
        (visual-fill-column-mode -1))
      (setq toa/splash--svg-rows 0)
      ;; GUI: SVG + wordmark. Terminal: just the dimmed wordmark.
      (if (display-graphic-p)
          (progn
            (toa/splash-insert-svg)
            (toa/splash-insert-centered toa/splash-text))
        (toa/splash-insert-centered toa/splash-text 'toa/splash-face))
      (insert "\n")
      (toa/splash-insert-centered
       (format "Loaded %d packages in %.2fs"
               (length package-activated-list)
               (float-time (time-subtract after-init-time before-init-time))))
      ;; Vertical centering: prepend blank lines. Count text rows plus the
      ;; image's extra rows so the block sits in the middle of the window.
      (let* ((content-rows (+ (count-lines (point-min) (point-max))
                              toa/splash--svg-rows))
             (top (max 0 (/ (- (window-body-height) content-rows) 2))))
        (goto-char (point-min))
        (insert (make-string top ?\n)))
      (goto-char (point-min))
      (read-only-mode 1))))

(defun toa/splash-on-resize (_frame)
  "Re-render splash screen when window size changes."
  (when (buffer-live-p (toa/splash-buffer))
    (dolist (win (get-buffer-window-list (toa/splash-buffer) nil t))
      (with-selected-window win
        (toa/splash-render)))))

(defun toa/splash-dismiss ()
  "Close the splash screen and remove resize hook."
  (interactive)
  (remove-hook 'window-size-change-functions #'toa/splash-on-resize)
  (when (buffer-live-p (toa/splash-buffer))
    (kill-buffer (toa/splash-buffer))))

(defun toa/splash-show ()
  "Show the splash screen (works in both GUI and terminal frames)."
  (interactive)
  (switch-to-buffer (toa/splash-buffer))
  (toa/splash-render)
  (add-hook 'window-size-change-functions #'toa/splash-on-resize)
  (use-local-map (let ((map (make-sparse-keymap)))
                   (define-key map (kbd "q") #'toa/splash-dismiss)
                   (define-key map (kbd "RET") #'toa/splash-dismiss)
                   (define-key map (kbd "SPC") #'toa/splash-dismiss)
                   map)))

;; Show the splash screen on startup, or on the first frame under a daemon.
(if (daemonp)
    (add-hook 'server-after-make-frame-hook
              (defun toa/splash-show-once ()
                (remove-hook 'server-after-make-frame-hook #'toa/splash-show-once)
                (toa/splash-show)))
  (add-hook 'emacs-startup-hook #'toa/splash-show))
#+end_src

;; Nag (once per session, only when something is missing) about the two
;; interactive setup steps that can't run unattended, echoing the exact command.
(defun toa/check-setup ()
  "Warn if nerd-icons fonts are missing."
  (let (todo)
    ;; Nerd Font glyphs: the modeline icons need this font installed OS-wide.
    (when (and (display-graphic-p)
               (require 'nerd-icons nil t)
               (not (find-font (font-spec :family nerd-icons-font-family))))
      (push "M-x nerd-icons-install-fonts   (then restart Emacs)" todo))
    (when todo
      (message "Setup needed:\n%s"
               (mapconcat (lambda (s) (concat "  • " s)) (nreverse todo) "\n")))))
